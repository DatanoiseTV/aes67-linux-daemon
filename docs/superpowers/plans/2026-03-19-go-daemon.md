# AES67 Go Daemon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task.

**Goal:** Rewrite the AES67 daemon in Go with direct netlink communication to the RAVENNA kernel module, serving the WebUI and full REST API.

**Architecture:** Go binary runs as root, opens netlink sockets, drops privileges. Packages: netlink (raw socket IPC), driver (kernel commands), session (source/sink management), config (JSON), api (HTTP server), discovery (SAP/mDNS), ptp (monitoring).

**Tech Stack:** Go 1.22+, standard library net/http, encoding/binary, syscall

**Spec:** `docs/superpowers/specs/2026-03-19-go-daemon-design.md`

### netlink.go — Raw netlink socket abstraction

```go
package netlink

import (
    "encoding/binary"
    "fmt"
    "sync"
    "syscall"
    "time"
    "unsafe"
)

const (
    ProtocolU2K  = 31 // User-to-Kernel (commands)
    ProtocolK2U  = 29 // Kernel-to-User (events)
    MaxPayload   = 1024
    ReplyTimeout = 1 * time.Second
)

type Conn struct {
    fd  int
    pid uint32
    mu  sync.Mutex
}

func Dial(protocol int) (*Conn, error) {
    fd, err := syscall.Socket(syscall.AF_NETLINK, syscall.SOCK_RAW, protocol)
    if err != nil { return nil, fmt.Errorf("netlink socket: %w", err) }

    sa := &syscall.SockaddrNetlink{
        Family: syscall.AF_NETLINK,
        Pid:    uint32(syscall.Getpid()),
    }
    if err := syscall.Bind(fd, sa); err != nil {
        syscall.Close(fd)
        return nil, fmt.Errorf("netlink bind: %w", err)
    }
    return &Conn{fd: fd, pid: uint32(syscall.Getpid())}, nil
}

func (c *Conn) Close() error { return syscall.Close(c.fd) }

// SendCommand sends a message and waits for response (U2K pattern)
func (c *Conn) SendCommand(msgID int32, payload []byte) (int32, []byte, error) {
    c.mu.Lock()
    defer c.mu.Unlock()

    // Build: nlmsghdr + MT_ALSA_msg + payload
    msg := buildMessage(c.pid, msgID, payload)

    // Send to kernel (pid=0)
    dst := &syscall.SockaddrNetlink{Family: syscall.AF_NETLINK, Pid: 0}
    if err := syscall.Sendto(c.fd, msg, 0, dst); err != nil {
        return 0, nil, fmt.Errorf("netlink send: %w", err)
    }

    // Set read deadline
    tv := syscall.Timeval{Sec: 1, Usec: 0}
    syscall.SetsockoptTimeval(c.fd, syscall.SOL_SOCKET, syscall.SO_RCVTIMEO, &tv)

    // Read response
    buf := make([]byte, 4096)
    n, _, err := syscall.Recvfrom(c.fd, buf, 0)
    if err != nil { return 0, nil, fmt.Errorf("netlink recv: %w", err) }

    // Parse response
    errCode, respPayload, err := parseResponse(buf[:n])
    return errCode, respPayload, err
}

// Listen blocks and receives events (K2U pattern), calls handler for each
func (c *Conn) Listen(handler func(msgID int32, payload []byte) []byte) error {
    buf := make([]byte, 4096)
    for {
        tv := syscall.Timeval{Sec: 1, Usec: 0}
        syscall.SetsockoptTimeval(c.fd, syscall.SOL_SOCKET, syscall.SO_RCVTIMEO, &tv)

        n, from, err := syscall.Recvfrom(c.fd, buf, 0)
        if err != nil {
            if isTimeout(err) { continue }
            return err
        }

        msgID, payload, err := parseEvent(buf[:n])
        if err != nil { continue }

        resp := handler(msgID, payload)
        if resp != nil {
            // Send response back to kernel
            syscall.Sendto(c.fd, resp, 0, from)
        }
    }
}
```

Implement helper functions: `buildMessage`, `parseResponse`, `parseEvent`, `isTimeout`.

### main.go
```go
func main() {
    // Parse flags
    configPath := flag.String("c", "/etc/daemon.conf", "config file")
    httpAddr := flag.String("a", "", "HTTP bind address")
    flag.Parse()

    // Load config
    cfg := config.Load(*configPath)

    // Open netlink sockets (requires root)
    drv, err := driver.New()

    // Drop privileges
    privileges.Drop("aes67-daemon")

    // Initialize components
    ptpMon := ptp.NewMonitor(drv)
    sessMgr := session.NewManager(drv, cfg, ptpMon)
    disc := discovery.New(cfg, sessMgr)
    srv := api.NewServer(cfg, sessMgr, ptpMon, disc)

    // Start
    drv.Hello(); drv.Start()
    sessMgr.Init() // load status.json, restore streams
    go ptpMon.Start(ctx)
    go disc.Start(ctx)
    go drv.ListenEvents(ctx)

    // Systemd notify ready
    daemon.SdNotify(false, daemon.SdNotifyReady)

    // Serve HTTP
    srv.ListenAndServe(cfg.HTTPPort)

    // Graceful shutdown on SIGTERM
}
```

### privileges.go
```go
func Drop(username string) error {
    u, _ := user.Lookup(username)
    uid, _ := strconv.Atoi(u.Uid)
    gid, _ := strconv.Atoi(u.Gid)
    syscall.Setgid(gid)
    syscall.Setuid(uid)
    return nil
}
```
